% Shared PGFPlots visual language for IEWC empirical figures. Figures are drawn
% inline at their final width, so these font sizes hold in every figure.
\definecolor{IEWCPrimary}{HTML}{3F21B6}
\definecolor{IEWCSecondary}{HTML}{8C7AD3}
\definecolor{IEWCDark}{HTML}{3F4147}
\definecolor{IEWCMid}{HTML}{777A80}
\definecolor{IEWCLight}{HTML}{B9BBC0}
\definecolor{IEWCText}{HTML}{20232A}
\definecolor{IEWCGrid}{HTML}{E3E5E8}
\definecolor{IEWCBand}{HTML}{F0F1F3}

\pgfplotsset{
  compat=1.18,
  iewc axis/.style={
    axis x line*=bottom,
    axis y line*=left,
    axis line style={IEWCText,line width=.45pt},
    tick align=outside,
    tick style={IEWCText,line width=.4pt},
    tick label style={font=\scriptsize,text=IEWCText},
    yticklabel style={/pgf/number format/fixed,/pgf/number format/precision=2},
    scale only axis,
    label style={font=\footnotesize,text=IEWCText},
    title style={font=\footnotesize,text=IEWCText,anchor=west,at={(0,1.02)}},
    ymajorgrids,
    grid style={IEWCGrid,line width=.35pt},
    legend style={draw=none,fill=none,font=\footnotesize,/tikz/every even column/.append style={column sep=8pt}},
    legend cell align=left,
    line width=.7pt,
    mark size=1.15pt,
    scaled ticks=false,
    clip=true,
  },
  sequential/.style={color=IEWCLight,mark=*,solid},
  ewc/.style={color=IEWCDark,mark=square*,solid},
  ewcdr/.style={color=IEWCMid,mark=triangle*,dotted},
  iewc/.style={color=IEWCPrimary,mark=*,solid,line width=.85pt},
  iewcrdw/.style={color=IEWCSecondary,mark=diamond*,dashed,line width=.8pt},
  context/.style={color=IEWCLight,mark=none,solid,line width=.6pt},
  target/.style={color=black,mark=none,solid,line width=.85pt},
  accuracy/.style={color=IEWCPrimary,mark=*,solid,line width=.85pt},
  retention/.style={color=IEWCSecondary,mark=square*,dashed,line width=.8pt},
  domainone/.style={color=IEWCPrimary,mark=*,solid,line width=.85pt},
  domaintwo/.style={color=IEWCSecondary,mark=diamond*,dashed,line width=.8pt},
}
